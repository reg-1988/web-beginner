前端监控

没有监控就没有优化的方向——错误监控、性能监控和行为埋点，是保障线上质量和用户体验的三驾马车。

1. 核心概念与定义

前端监控是收集、分析和展示前端应用运行时数据的过程，包括错误监控（JS 错误、资源加载失败、接口异常）、性能监控（页面加载、渲染、交互指标）和行为监控（用户操作、页面访问、功能使用）。

监控的价值：发现问题（线上 Bug 实时感知）、定位问题（错误堆栈 + 用户环境）、度量效果（优化前后数据对比）、辅助决策（用户行为数据驱动产品）。

2. 关键特性与工作原理

2.1 三大监控类型

类型监控内容采集方式
错误监控JS 异常、Promise 拒绝、资源加载失败、接口错误window.onerror, unhandledrejection, 资源监听
性能监控FCP/LCP/CLS/INP、资源加载、接口耗时Performance API, PerformanceObserver
行为监控页面 PV/UV、点击、路由跳转、功能使用事件监听、路由 Hook、手动埋点

2.2 数据上报方式

方式特点适用场景
navigator.sendBeacon异步、页面卸载时不丢失关闭/跳转时上报
fetch (keepalive)类似 sendBeacon，更灵活现代浏览器
1x1 GIF 图片无跨域限制、兼容性好兼容性方案
XMLHttpRequest同步可能丢失不推荐

示例一：错误监控 SDK

JavaScriptclass ErrorMonitor {
  constructor(reportUrl) {
    this.reportUrl = reportUrl;
    this.queue = [];
  }

  init() {
    // JS 运行时错误
    window.onerror = (msg, url, line, col, error) => {
      this.report({
        type: 'js_error',
        message: msg,
        filename: url,
        line, col,
        stack: error?.stack,
      });
    };

    // Promise 未捕获拒绝
    window.addEventListener('unhandledrejection', (e) => {
      this.report({
        type: 'promise_error',
        message: e.reason?.message || String(e.reason),
        stack: e.reason?.stack,
      });
    });

    // 资源加载错误（冒泡到 window）
    window.addEventListener('error', (e) => {
      if (e.target !== window) {
        this.report({
          type: 'resource_error',
          tagName: e.target.tagName,
          src: e.target.src || e.target.href,
        });
      }
    }, true); // 捕获阶段
  }

  report(data) {
    const payload = {
      ...data,
      url: location.href,
      timestamp: Date.now(),
      userAgent: navigator.userAgent,
    };
    this.queue.push(payload);
    this.flush();
  }

  flush() {
    if (this.flushing || this.queue.length === 0) return;
    this.flushing = true;
    const batch = this.queue.splice(0, 10);
    navigator.sendBeacon(
      this.reportUrl,
      JSON.stringify(batch)
    );
    this.flushing = false;
    if (this.queue.length) setTimeout(() => this.flush(), 1000);
  }
}

示例二：性能指标采集

JavaScriptclass PerfMonitor {
  constructor(reportUrl) {
    this.reportUrl = reportUrl;
    this.metrics = {};
  }

  init() {
    // 导航性能
    const nav = performance.getEntriesByType('navigation')[0];
    if (nav) {
      this.metrics.dns = nav.domainLookupEnd - nav.domainLookupStart;
      this.metrics.tcp = nav.connectEnd - nav.connectStart;
      this.metrics.ttfb = nav.responseStart - nav.requestStart;
      this.metrics.domReady = nav.domContentLoadedEventEnd - nav.startTime;
      this.metrics.load = nav.loadEventEnd - nav.startTime;
    }

    // Web Vitals
    this.observe('paint', (entry) => {
      this.metrics[entry.name] = entry.startTime; // first-paint, first-contentful-paint
    });
    this.observe('largest-contentful-paint', (entry) => {
      this.metrics.LCP = entry.renderTime || entry.loadTime;
    });
    this.observe('layout-shift', (entry) => {
      if (!entry.hadRecentInput) {
        this.metrics.CLS = (this.metrics.CLS || 0) + entry.value;
      }
    });
    this.observe('first-input', (entry) => {
      this.metrics.FID = entry.processingStart - entry.startTime;
    });

    // 页面卸载时上报
    window.addEventListener('visibilitychange', () => {
      if (document.visibilityState === 'hidden') this.report();
    });
  }

  observe(type, callback) {
    try {
      new PerformanceObserver((list) => {
        list.getEntries().forEach(callback);
      }).observe({ type, buffered: true });
    } catch {}
  }

  report() {
    navigator.sendBeacon(this.reportUrl, JSON.stringify({
      type: 'performance',
      metrics: this.metrics,
      url: location.href,
      timestamp: Date.now(),
    }));
  }
}

示例三：声明式埋点方案

HTML<!-- 声明式埋点：通过 data 属性标记 -->
<button data-track="click_login" data-track-params='{"method":"wechat"}'>
  微信登录
</button>
<div data-track="expose_banner" data-track-params='{"id":"summer_sale"}'>
  活动 Banner
</div>
JavaScript// 自动采集点击事件
document.addEventListener('click', (e) => {
  const target = e.target.closest('[data-track]');
  if (!target) return;
  
  const eventName = target.dataset.track;
  const params = JSON.parse(target.dataset.trackParams || '{}');
  
  track(eventName, params);
});

// 曝光采集（IntersectionObserver）
function setupExposeTracking() {
  const observer = new IntersectionObserver((entries) => {
    entries.forEach(entry => {
      if (entry.isIntersecting) {
        const el = entry.target;
        const eventName = el.dataset.track;
        const params = JSON.parse(el.dataset.trackParams || '{}');
        track(`expose_${eventName}`, params);
        observer.unobserve(el); // 只上报一次
      }
    });
  }, { threshold: 0.5 }); // 50% 可见时触发

  document.querySelectorAll('[data-track^="expose_"]').forEach(el => {
    observer.observe(el);
  });
}

// 埋点上报函数
function track(event, params = {}) {
  navigator.sendBeacon('/api/track', JSON.stringify({
    event,
    params,
    url: location.href,
    referrer: document.referrer,
    timestamp: Date.now(),
    userId: getUserId(),
    sessionId: getSessionId(),
  }));
}

示例四：SourceMap 解析

JavaScript// 线上代码压缩后错误堆栈不可读
// 需要 SourceMap 还原原始位置

// 方案：服务端解析（推荐，SourceMap 不暴露给用户）
// 1. 构建时生成 SourceMap，上传到监控服务器
// 2. 线上不部署 SourceMap（安全）
// 3. 错误上报时包含压缩后的堆栈
// 4. 监控服务用 source-map 库解析还原

const { SourceMapConsumer } = require('source-map');

async function resolveStackTrace(rawStack, sourceMapContent) {
  const consumer = await new SourceMapConsumer(sourceMapContent);
  
  const parsed = parseStackTrace(rawStack); // 解析为帧数组
  const resolved = parsed.map(frame => {
    const pos = consumer.originalPositionFor({
      line: frame.line,
      column: frame.column,
    });
    return {
      ...frame,
      source: pos.source,   // 原始文件名
      line: pos.line,        // 原始行号
      column: pos.column,    // 原始列号
      name: pos.name,        // 原始变量名
    };
  });
  
  consumer.destroy();
  return resolved;
}

   - 核心 API 可用性 < 99.9% → P0 告警

4. 常见问题与坑点

SourceMap 暴露在线上安全吗？ +

不安全。SourceMap 包含完整源码，用户可以在 DevTools 中看到。解决方案：① 线上不部署 SourceMap；② 构建时生成 SourceMap 并上传到监控服务；③ 设置服务器仅允许内网 IP 访问 SourceMap；④ 使用隐藏的 HTTP Header 控制访问。

监控数据量太大怎么办？ +

① 采样上报（如只上报 10% 的 PV 数据）；② 聚合上报（批量合并、1 秒内合并为一条）；③ 限频（同一错误 1 分钟内只上报一次）；④ 客户端过滤（忽略已知第三方脚本错误）；⑤ 数据降级（非核心指标降低精度）。

🚫 不要在监控中收集敏感信息

错误堆栈、URL 参数、请求体中可能包含用户密码、Token 等敏感信息。监控 SDK 应在采集前脱敏处理：过滤 Authorization 头、密码字段、URL 中的 token 参数等。
