H5 性能优化

移动端 H5 面临网络慢、内存少、CPU 弱的挑战——首屏优化、懒加载、虚拟列表和骨架屏是核心优化手段。

1. 核心概念与定义

移动端 H5 性能优化的核心目标：首屏快速呈现（1~2 秒内可见内容）和交互流畅（60fps 不卡顿）。移动端相比桌面端有更多限制：网络延迟高、CPU/GPU 性能弱、内存有限、电量敏感。

优化分为两大方向：加载优化（减少资源体积、加快传输速度、优化加载策略）和运行时优化（减少重排重绘、优化 JS 执行、内存管理）。

示例一：虚拟列表实现

JavaScript/**
 * 虚拟列表：只渲染可见区域的 DOM 节点
 * 适合渲染成千上万条数据
 */
class VirtualList {
  constructor(container, options = {}) {
    this.container = container;
    this.itemHeight = options.itemHeight || 50;
    this.buffer = options.buffer || 5; // 缓冲区
    this.data = [];
    this.visibleItems = [];
    
    this.scrollEl = document.createElement('div');
    this.scrollEl.style.overflow = 'auto';
    this.scrollEl.style.height = '100%';
    
    this.contentEl = document.createElement('div');
    this.contentEl.style.position = 'relative';
    
    this.scrollEl.appendChild(this.contentEl);
    container.appendChild(this.scrollEl);
    
    this.scrollEl.addEventListener('scroll', () => this.render());
  }

  setData(data) {
    this.data = data;
    this.contentEl.style.height = `${data.length * this.itemHeight}px`;
    this.render();
  }

  render() {
    const scrollTop = this.scrollEl.scrollTop;
    const viewHeight = this.scrollEl.clientHeight;
    
    const startIdx = Math.max(0, Math.floor(scrollTop / this.itemHeight) - this.buffer);
    const endIdx = Math.min(
      this.data.length,
      Math.ceil((scrollTop + viewHeight) / this.itemHeight) + this.buffer
    );
    
    // 清空并重新渲染可见区域
    this.contentEl.innerHTML = '';
    for (let i = startIdx; i < endIdx; i++) {
      const item = this.createItem(this.data[i], i);
      item.style.position = 'absolute';
      item.style.top = `${i * this.itemHeight}px`;
      item.style.height = `${this.itemHeight}px`;
      item.style.width = '100%';
      this.contentEl.appendChild(item);
    }
  }

  createItem(data, index) {
    const el = document.createElement('div');
    el.textContent = `Item ${index}: ${data.name}`;
    el.style.padding = '0 16px';
    el.style.lineHeight = `${this.itemHeight}px`;
    el.style.boxSizing = 'border-box';
    el.style.borderBottom = '1px solid #eee';
    return el;
  }
}

示例二：图片懒加载

JavaScript// IntersectionObserver 方案（推荐）
function setupLazyImages() {
  const images = document.querySelectorAll('img[data-src]');
  
  const observer = new IntersectionObserver((entries) => {
    entries.forEach(entry => {
      if (entry.isIntersecting) {
        const img = entry.target;
        img.src = img.dataset.src;
        img.removeAttribute('data-src');
        observer.unobserve(img);
      }
    });
  }, {
    rootMargin: '200px', // 提前 200px 开始加载
  });

  images.forEach(img => observer.observe(img));
}

示例五：性能监控 SDK

JavaScript/**
 * 简易前端性能监控
 */
class PerfMonitor {
  constructor(reportUrl) {
    this.reportUrl = reportUrl;
    this.metrics = {};
  }

  init() {
    this.observePaint();
    this.observeLCP();
    this.observeFID();
    this.observeCLS();
    this.observeResourceTiming();
    window.addEventListener('load', () => this.report());
  }

  observePaint() {
    const observer = new PerformanceObserver((list) => {
      for (const entry of list.getEntries()) {
        this.metrics[entry.name] = entry.startTime;
      }
    });
    observer.observe({ type: 'paint', buffered: true });
  }

  observeLCP() {
    const observer = new PerformanceObserver((list) => {
      const entries = list.getEntries();
      this.metrics.LCP = entries[entries.length - 1].renderTime;
    });
    observer.observe({ type: 'largest-contentful-paint', buffered: true });
  }

  observeFID() {
    const observer = new PerformanceObserver((list) => {
      for (const entry of list.getEntries()) {
        this.metrics.FID = entry.processingStart - entry.startTime;
      }
    });
    observer.observe({ type: 'first-input', buffered: true });
  }

  observeCLS() {
    let clsValue = 0;
    const observer = new PerformanceObserver((list) => {
      for (const entry of list.getEntries()) {
        if (!entry.hadRecentInput) clsValue += entry.value;
      }
      this.metrics.CLS = clsValue;
    });
    observer.observe({ type: 'layout-shift', buffered: true });
  }

  observeResourceTiming() {
    const observer = new PerformanceObserver((list) => {
      const resources = list.getEntries()
        .filter(e => e.initiatorType === 'xmlhttprequest')
        .map(e => ({
          url: e.name,
          duration: e.duration,
          size: e.transferSize,
        }));
      this.metrics.slowApis = resources.filter(r => r.duration > 3000);
    });
    observer.observe({ type: 'resource', buffered: true });
  }

  report() {
    setTimeout(() => {
      navigator.sendBeacon(this.reportUrl, JSON.stringify({
        url: location.href,
        metrics: this.metrics,
        timestamp: Date.now(),
      }));
    }, 3000); // 延迟上报，确保指标完整
  }
}

4. 常见问题与坑点

虚拟列表在什么场景下使用？ +

当列表项超过 100 条且每项 DOM 复杂度高时考虑使用。判断标准：如果列表渲染导致页面卡顿（掉帧），就需要虚拟列表。少于 50 条的简单列表不需要，直接渲染性能更好。

骨架屏和 Loading 圈哪个好？ +

骨架屏更好。研究表明骨架屏降低了用户感知的等待时间。原因：骨架屏提供了页面结构预期，让用户知道内容即将出现；Loading 圈只传达"正在等待"，没有结构预期。推荐骨架屏 + 渐进式加载的组合。

🚫 移动端慎用大图和动画

移动端 GPU 性能弱，大图（> 1MB）和高频动画容易导致内存溢出和卡顿。图片务必压缩、使用 WebP、设置合理的 max-width。动画只用 transform/opacity，避免触发重排。

2虚拟列表的原理？
只渲染可视区域内的 DOM 节点。计算 scrollTop 和容器高度确定可见范围，动态创建/销毁 DOM。用绝对定位或 transform 控制位置。内容区域设置总高度撑起滚动条。滚动时重新计算可见范围并更新 DOM。
